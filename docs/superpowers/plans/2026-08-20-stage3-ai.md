# KnowDo Stage 3 (AI) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upload documents to the Go API, have a Python worker ingest them asynchronously into Qdrant, and answer natural-language questions about them via RAG.

**Architecture:** The Go API stays the only publicly-routed service. It stores uploaded bytes in Postgres and pushes an ingest job onto a Redis list. A Python worker image runs as two Compose services from one build: a `BLPOP` consumer that extracts/chunks/embeds/upserts, and a FastAPI service exposing `POST /ask`. Go's `POST /ask` proxies to that service synchronously.

**Tech Stack:** Go 1.25 (`net/http`, `pgx/v5`, `go-redis/v9`), Python 3.12 (FastAPI, `fastembed`, `qdrant-client`, `psycopg`, `openai` SDK, `pypdf`), Postgres 16, Redis 7, Qdrant, Docker Compose.

**Spec:** `docs/superpowers/specs/2026-08-20-stage3-ai-design.md`

## Global Constraints

- Go module is `knowdo/api`, rooted at `api/`. Go 1.25.0 per `api/go.mod`.
- Go tests use stdlib `testing` + `net/http/httptest`, table-driven. No assertion library.
- Handlers depend on a `Store` interface, never on `*PGStore`. Same for the queue and the AI client. This is what Stage 1 established; follow it.
- No ORM. Direct SQL through `pgx`.
- Migrations are `golang-migrate` `.sql` pairs in `api/migrations/`, applied via the `migrate` CLI. Never auto-applied at startup.
- Python lives in `worker/`, Python 3.12, linted with `ruff`, tested with `pytest`.
- **No test may touch the network.** Not the embedding model download, not the LLM API, not a real Qdrant server. `qdrant-client` local mode (`QdrantClient(":memory:")`) is a real Qdrant implementation and is allowed.
- LLM configuration is provider-neutral: `LLM_BASE_URL`, `LLM_MODEL`, `LLM_API_KEY`. Never name a vendor in code.
- Redis list key is exactly `knowdo:jobs:ingest`. Job payload is exactly `{"document_id": <int>}`.
- Qdrant collection is exactly `knowdo`, 384 dimensions, cosine distance.
- Document status is one of exactly: `pending`, `processing`, `ready`, `failed`.
- Deliberate simplifications get a `ponytail:` comment naming the ceiling and the upgrade path.
- Commit after every task. Do not commit with failing tests.

package document

import (
	"context"
	"errors"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

var ErrNotFound = errors.New("document not found")

// Store is the persistence boundary for documents. Handlers depend on
// this interface, not on *PGStore directly, so tests can swap in a fake.
type Store interface {
	Create(ctx context.Context, d *Document, content []byte) error
	List(ctx context.Context) ([]Document, error)
	Get(ctx context.Context, id int64) (Document, error)
}

type PGStore struct {
	pool *pgxpool.Pool
}

func NewPGStore(pool *pgxpool.Pool) *PGStore {
	return &PGStore{pool: pool}
}

// Create inserts the metadata and the file bytes together, filling in
// the DB-generated fields from the RETURNING clause.
func (s *PGStore) Create(ctx context.Context, d *Document, content []byte) error {
	const q = `
		INSERT INTO documents (filename, content_type, content)
		VALUES ($1, $2, $3)
		RETURNING id, status, chunk_count, created_at`

	return s.pool.QueryRow(ctx, q, d.Filename, d.ContentType, content).
		Scan(&d.ID, &d.Status, &d.ChunkCount, &d.CreatedAt)
}

// List never selects content — a list endpoint that loads every
// uploaded file into memory is a trivially avoidable failure.
func (s *PGStore) List(ctx context.Context) ([]Document, error) {
	const q = `
		SELECT id, filename, content_type, status, error, chunk_count, created_at
		FROM documents ORDER BY id`

	rows, err := s.pool.Query(ctx, q)
	if err != nil {
		return nil, err
	}

	defer rows.Close()

	docs := []Document{}
	for rows.Next() {
		var d Document

		if err := rows.Scan(&d.ID, &d.Filename, &d.ContentType, &d.Status, &d.Error, &d.ChunkCount, &d.CreatedAt); err != nil {
			return nil, err
		}

		docs = append(docs, d)
	}

	return docs, rows.Err()
}

func (s *PGStore) Get(ctx context.Context, id int64) (Document, error) {
	const q = `
		SELECT id, filename, content_type, status, error, chunk_count, created_at
		FROM documents WHERE id = $1`

	var d Document

	err := s.pool.QueryRow(ctx, q, id).
		Scan(&d.ID, &d.Filename, &d.ContentType, &d.Status, &d.Error, &d.ChunkCount, &d.CreatedAt)

	if errors.Is(err, pgx.ErrNoRows) {
		return Document{}, ErrNotFound
	}

	return d, err
}
```

// Package queue pushes background jobs onto Redis lists for the Python
// worker to consume.
package queue

import (
	"context"
	"encoding/json"

	"github.com/redis/go-redis/v9"
)

// IngestList is the Redis list the Python consumer BLPOPs from. The
// name is part of the cross-service contract; changing it here means
// changing it in worker/knowdo_worker/consumer.py too.
const IngestList = "knowdo:jobs:ingest"

// IngestJob is the payload the worker receives.
type IngestJob struct {
	DocumentID int64 `json:"document_id"`
}

func (j IngestJob) payload() ([]byte, error) {
	return json.Marshal(j)
}

// Queue is the enqueue boundary. Handlers depend on this interface so
// tests don't need a running Redis.
type Queue interface {
	EnqueueIngest(ctx context.Context, documentID int64) error
}

type Redis struct {
	client *redis.Client
}

func NewRedis(url string) (*Redis, error) {
	opts, err := redis.ParseURL(url)
	if err != nil {
		return nil, err
	}

	return &Redis{client: redis.NewClient(opts)}, nil
}

func (q *Redis) EnqueueIngest(ctx context.Context, documentID int64) error {
	body, err := IngestJob{DocumentID: documentID}.payload()
	if err != nil {
		return err
	}

	// ponytail: plain LPUSH, no retry/DLQ. A dropped job leaves the
	// document at status 'pending' and is re-driven by re-uploading.
	// Add a reliable queue (BRPOPLPUSH + ack list) when a job is
	// actually observed to be lost.
	return q.client.LPush(ctx, IngestList, body).Err()
}

func (q *Redis) Ping(ctx context.Context) error {
	return q.client.Ping(ctx).Err()
}
```

package document

import (
	"encoding/json"
	"errors"
	"io"
	"log"
	"net/http"
	"strconv"

	"knowdo/api/internal/queue"
)

// maxUploadBytes caps an upload because Create reads the whole file into
// memory before inserting it. ponytail: in-memory read, 10MB ceiling —
// switch to a streaming large-object write if real documents outgrow it.
const maxUploadBytes = 10 << 20

type Handlers struct {
	store Store
	queue queue.Queue
}

func NewHandlers(store Store, q queue.Queue) *Handlers {
	return &Handlers{store: store, queue: q}
}

// HandleCreate accepts a multipart upload, stores it as 'pending', and
// queues the ingest job. It returns 202, not 201: the document exists,
// but it is not yet answerable — the client polls GET /documents/{id}
// for the status to reach 'ready'.
func (h *Handlers) HandleCreate(w http.ResponseWriter, r *http.Request) {
	r.Body = http.MaxBytesReader(w, r.Body, maxUploadBytes)

	file, header, err := r.FormFile("file")
	if err != nil {
		http.Error(w, "a 'file' upload is required", http.StatusBadRequest)
		return
	}

	defer file.Close()

	content, err := io.ReadAll(file)
	if err != nil {
		http.Error(w, "failed to read upload", http.StatusBadRequest)
		return
	}

	if len(content) == 0 {
		http.Error(w, "uploaded file is empty", http.StatusBadRequest)
		return
	}

	d := &Document{
		Filename:    header.Filename,
		ContentType: header.Header.Get("Content-Type"),
	}

	if err := h.store.Create(r.Context(), d, content); err != nil {
		http.Error(w, "failed to store document", http.StatusInternalServerError)
		return
	}

	// A stored-but-unqueued document would sit at 'pending' forever
	// with no worker coming, so a failed enqueue is a failed request.
	if err := h.queue.EnqueueIngest(r.Context(), d.ID); err != nil {
		log.Printf("enqueueing ingest job for document %d: %v", d.ID, err)
		http.Error(w, "failed to queue document for processing", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusAccepted)
	if err := json.NewEncoder(w).Encode(d); err != nil {
		log.Printf("encoding response: %v", err)
	}
}

func (h *Handlers) HandleList(w http.ResponseWriter, r *http.Request) {
	docs, err := h.store.List(r.Context())
	if err != nil {
		http.Error(w, "failed to list documents", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	if err := json.NewEncoder(w).Encode(docs); err != nil {
		log.Printf("encoding response: %v", err)
	}
}

func (h *Handlers) HandleGet(w http.ResponseWriter, r *http.Request) {
	id, err := strconv.ParseInt(r.PathValue("id"), 10, 64)
	if err != nil {
		http.Error(w, "invalid id", http.StatusBadRequest)
		return
	}

	d, err := h.store.Get(r.Context(), id)
	if err != nil {
		if errors.Is(err, ErrNotFound) {
			http.Error(w, "document not found", http.StatusNotFound)
			return
		}

		http.Error(w, "internal error", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	if err := json.NewEncoder(w).Encode(d); err != nil {
		log.Printf("encoding response: %v", err)
	}
}

import (
	"encoding/json"

func NewRouter(pool *pgxpool.Pool, q queue.Queue) *http.ServeMux {
	mux := http.NewServeMux()

	taskHandlers := task.NewHandlers(task.NewPGStore(pool))
	documentHandlers := document.NewHandlers(document.NewPGStore(pool), q)

	mux.HandleFunc("GET /health", handleHealth)
	mux.HandleFunc("POST /tasks", taskHandlers.HandleCreate)
	mux.HandleFunc("GET /tasks", taskHandlers.HandleList)
	mux.HandleFunc("GET /tasks/{id}", taskHandlers.HandleGet)
	mux.HandleFunc("PATCH /tasks/{id}", taskHandlers.HandleUpdate)
	mux.HandleFunc("DELETE /tasks/{id}", taskHandlers.HandleDelete)
	mux.HandleFunc("POST /documents", documentHandlers.HandleCreate)
	mux.HandleFunc("GET /documents", documentHandlers.HandleList)
	mux.HandleFunc("GET /documents/{id}", documentHandlers.HandleGet)
	mux.Handle("/", http.FileServer(http.Dir("static")))

	return mux

```python
"""Split extracted document text into embeddable pieces."""


def chunk_text(text: str, size: int = 800, overlap: int = 100) -> list[str]:
    """Split text into overlapping fixed-width windows.

    ponytail: a naive character splitter — it cuts mid-word and mid-
    sentence, and 800 chars sits comfortably under bge-small's 512-token
    limit. The overlap is what keeps a fact spanning a boundary
    retrievable. Upgrade to a token-aware or sentence-aware splitter only
    if retrieval quality visibly suffers.
    """
    if overlap >= size:
        raise ValueError(f"overlap ({overlap}) must be smaller than size ({size})")

    text = text.strip()
    if not text:
        return []

    step = size - overlap
    return [text[i : i + size] for i in range(0, len(text), step) if text[i : i + size]]
```

```python
"""Turn uploaded bytes into plain text."""

import io

from pypdf import PdfReader


class ExtractionError(Exception):
    """Raised when bytes cannot be turned into usable text.

    The message reaches the user through documents.error, so write it
    for a human reading it in the UI, not for a log grep.
    """


def extract_text(content: bytes, filename: str, content_type: str = "") -> str:
    """Extract text from an uploaded document.

    ponytail: PDF and UTF-8 text only. DOCX, HTML and OCR for scanned
    PDFs are each a dependency, and none is needed until someone
    actually uploads one.
    """
    if _is_pdf(content, filename, content_type):
        return _extract_pdf(content)

    return _extract_plain(content)


def _is_pdf(content: bytes, filename: str, content_type: str) -> bool:
    # Magic bytes first: content types are unreliable and users rename
    # files, but a PDF always starts with %PDF-.
    return (
        content.startswith(b"%PDF-")
        or content_type == "application/pdf"
        or filename.lower().endswith(".pdf")
    )


def _extract_pdf(content: bytes) -> str:
    try:
        reader = PdfReader(io.BytesIO(content))
        text = "\n".join(page.extract_text() or "" for page in reader.pages)
    except Exception as exc:
        raise ExtractionError(f"could not read PDF: {exc}") from exc

    if not text.strip():
        raise ExtractionError(
            "PDF contains no extractable text (it may be a scan; OCR is not supported)"
        )

    return text


def _extract_plain(content: bytes) -> str:
    try:
        text = content.decode("utf-8")
    except UnicodeDecodeError as exc:
        raise ExtractionError("file is not a PDF and is not valid UTF-8 text") from exc

    if not text.strip():
        raise ExtractionError("file contains no text")

    return text
```

Embeddings run locally rather than through the LLM provider: it is
cheaper, works offline, and keeps the test suite free of network calls.
fastembed is ONNX-based, so this does not pull in PyTorch.
"""

import os

from fastembed import TextEmbedding

EMBED_DIM = 384

_model: TextEmbedding | None = None

    # Loaded lazily and cached: importing this module must stay cheap so
    # tests can import its siblings without paying for a model load.
    global _model

    if _model is None:
        _model = TextEmbedding(model_name=EMBED_MODEL)

    return _model


def embed(texts: list[str]) -> list[list[float]]:
    """Embed a batch of texts into EMBED_DIM-dimensional vectors."""
    return [vector.tolist() for vector in _get_model().embed(texts)]

```python
"""Qdrant collection management, upserts and similarity search."""

import os

from qdrant_client import QdrantClient, models

from knowdo_worker.embed import EMBED_DIM

COLLECTION = "knowdo"

# Chunk point ids are derived as document_id * CHUNK_ID_STRIDE +
# chunk_index so re-ingesting a document overwrites its own chunks and
# never another document's. ponytail: this caps a document at 100k
# chunks (~80MB of text); switch to UUIDs if that is ever a real limit.
CHUNK_ID_STRIDE = 100_000


def get_client() -> QdrantClient:
    return QdrantClient(url=os.getenv("QDRANT_URL", "http://qdrant:6333"))


def ensure_collection(client: QdrantClient) -> None:
    """Create the collection if it does not exist. Safe to call on every start."""
    if client.collection_exists(COLLECTION):
        return

    client.create_collection(
        collection_name=COLLECTION,
        vectors_config=models.VectorParams(size=EMBED_DIM, distance=models.Distance.COSINE),
    )


def upsert_chunks(
    client: QdrantClient,
    document_id: int,
    chunks: list[str],
    vectors: list[list[float]],
) -> None:
    """Store a document's chunks and their vectors."""
    client.upsert(
        collection_name=COLLECTION,
        points=[
            models.PointStruct(
                id=document_id * CHUNK_ID_STRIDE + index,
                vector=vector,
                payload={"document_id": document_id, "chunk_index": index, "text": text},
            )
            for index, (text, vector) in enumerate(zip(chunks, vectors, strict=True))
        ],
    )


def search(client: QdrantClient, vector: list[float], limit: int = 5) -> list[dict]:
    """Return the most similar chunks as plain dicts."""
    result = client.query_points(
        collection_name=COLLECTION, query=vector, limit=limit, with_payload=True
    )

    return [
        {
            "document_id": point.payload["document_id"],
            "chunk_index": point.payload["chunk_index"],
            "text": point.payload["text"],
            "score": point.score,
        }
        for point in result.points
    ]
```

"""The two Postgres statements the worker needs."""

import os

import psycopg


class DocumentNotFound(Exception):
    pass


def connect() -> psycopg.Connection:
    return psycopg.connect(os.environ["DATABASE_URL"], autocommit=True)


def fetch_document(conn, document_id: int) -> tuple[str, str, bytes]:
    """Return (filename, content_type, content) for a document."""
    with conn.cursor() as cur:
        cur.execute(
            "SELECT filename, content_type, content FROM documents WHERE id = %s",
            (document_id,),
        )
        row = cur.fetchone()

    if row is None:
        raise DocumentNotFound(f"document {document_id} does not exist")

    return row[0], row[1], bytes(row[2])

    """Write back the outcome. This is how the API and the user learn
    that ingestion finished — a failure visible only in worker logs is a
    failure nobody sees."""
    with conn.cursor() as cur:
        cur.execute(
            "UPDATE documents SET status = %s, error = %s, chunk_count = %s WHERE id = %s",
            (status, error, chunk_count, document_id),

```python
"""The document ingest pipeline: extract -> chunk -> embed -> upsert."""

import logging

from knowdo_worker.chunk import chunk_text
from knowdo_worker.db import fetch_document, set_status
from knowdo_worker.embed import embed
from knowdo_worker.extract import ExtractionError, extract_text
from knowdo_worker.vectors import upsert_chunks

log = logging.getLogger(__name__)


def ingest(conn, client, document_id: int, embed_fn=embed) -> int:
    """Ingest one document, returning the number of chunks stored.

    embed_fn is injectable so tests can run the whole pipeline without
    downloading a model.

    On failure the document is marked 'failed' with a human-readable
    reason and the exception is re-raised for the caller to log.
    ponytail: no retries and no dead-letter queue — re-upload to retry.
    Add retries when a job is actually observed failing transiently.
    """
    filename, content_type, content = fetch_document(conn, document_id)
    set_status(conn, document_id, "processing")

    try:
        text = extract_text(content, filename, content_type)
        chunks = chunk_text(text)

        if not chunks:
            raise ExtractionError("document produced no text to index")

        upsert_chunks(client, document_id, chunks, embed_fn(chunks))
    except Exception as exc:
        log.exception("ingesting document %d failed", document_id)
        set_status(conn, document_id, "failed", error=str(exc))
        raise

    set_status(conn, document_id, "ready", chunk_count=len(chunks))
    return len(chunks)
```

```python
"""Redis queue consumer: BLPOP an ingest job, run the pipeline."""

import json
import logging
import os

import redis

from knowdo_worker.db import connect
from knowdo_worker.ingest import ingest
from knowdo_worker.vectors import ensure_collection, get_client

# Must match queue.IngestList in api/internal/queue/queue.go.
INGEST_LIST = "knowdo:jobs:ingest"

log = logging.getLogger(__name__)


def handle_job(payload: str, conn, client) -> None:
    """Process one job. Never raises: a single bad message must not stop
    the worker, or every document queued behind it would go unprocessed.
    Ingest failures are already recorded on the document itself."""
    try:
        job = json.loads(payload)
        document_id = job["document_id"]

        if not isinstance(document_id, int):
            raise TypeError(f"document_id must be an int, got {document_id!r}")
    except (ValueError, TypeError, KeyError):
        log.exception("skipping malformed job payload: %r", payload)
        return

    try:
        chunks = ingest(conn, client, document_id)
        log.info("ingested document %d into %d chunks", document_id, chunks)
    except Exception:
        log.exception("ingesting document %d failed", document_id)


def run() -> None:
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")

    client = get_client()
    ensure_collection(client)

    conn = connect()
    r = redis.from_url(os.environ["REDIS_URL"], decode_responses=True)

    log.info("consuming %s", INGEST_LIST)
    while True:
        _, payload = r.blpop(INGEST_LIST)
        handle_job(payload, conn, client)


if __name__ == "__main__":
    run()
```

"""The LLM boundary — one function, configured by environment.

The OpenAI chat-completions shape is the de-facto standard: Kimi,
DeepSeek, Groq, Together, OpenRouter, Ollama, vLLM and llama.cpp's
server all speak it. So swapping providers is three environment
variables, not a class hierarchy. If a genuinely non-compatible provider
is ever needed it becomes a branch inside chat(), with no caller
changes. Stage 4's LangChain will likely replace this file entirely,
which is another reason to keep it one function.
"""

import os

from openai import OpenAI

_client: OpenAI | None = None


def _get_client() -> OpenAI:
    global _client

    if _client is None:
        _client = OpenAI(
            base_url=os.environ["LLM_BASE_URL"],
            api_key=os.environ["LLM_API_KEY"],
        )

    return _client


def chat(messages: list[dict]) -> str:
    """Send a chat completion and return the assistant's text."""
    response = _get_client().chat.completions.create(
        model=os.environ["LLM_MODEL"],
        messages=messages,
        temperature=0.2,
    )

    return response.choices[0].message.content or ""

```python
"""The ask pipeline: embed the question, retrieve chunks, ask the model."""

from knowdo_worker.embed import embed
from knowdo_worker.llm import chat
from knowdo_worker.vectors import search

TOP_K = 5

SYSTEM_PROMPT = (
    "You answer questions about the user's own uploaded notes and documents. "
    "Use only the provided context. If the context does not contain the answer, "
    "say so plainly rather than guessing."
)

NO_CONTEXT_ANSWER = (
    "I could not find anything in your documents that relates to that question. "
    "Try uploading the relevant document first, or rephrasing the question."
)


def answer(question: str, client, embed_fn=embed, chat_fn=chat) -> dict:
    """Answer a question from stored document chunks.

    embed_fn and chat_fn are injectable so tests exercise the retrieval
    and prompt-building logic without a model download or an API call.

    ponytail: plain top-K stuffing — no reranking, no query rewriting, no
    conversation history. Add them if answers are visibly poor; Stage 4's
    workflow work is where history belongs.
    """
    query_vector = embed_fn([question])[0]
    hits = search(client, query_vector, limit=TOP_K)

    if not hits:
        # No context means the model can only hallucinate. Don't pay for that.
        return {"answer": NO_CONTEXT_ANSWER, "sources": []}

    context = "\n\n---\n\n".join(hit["text"] for hit in hits)
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": f"Context:\n\n{context}\n\nQuestion: {question}"},
    ]

    # dict.fromkeys keeps first-seen order, so sources stay ranked by relevance.
    sources = list(dict.fromkeys(hit["document_id"] for hit in hits))

    return {"answer": chat_fn(messages), "sources": sources}
```

"""FastAPI service exposing RAG queries to the Go API.

This service is internal: the Go API is the only publicly-routed
service, and it is where auth and rate limiting will live.
"""

import logging
from contextlib import asynccontextmanager

from fastapi import FastAPI
from pydantic import BaseModel, field_validator

from knowdo_worker.rag import answer
from knowdo_worker.vectors import ensure_collection, get_client

log = logging.getLogger(__name__)

# Set on startup; module-level so tests can substitute it.
_vectors = None


@asynccontextmanager
async def lifespan(app: FastAPI):
    global _vectors

    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
    _vectors = get_client()
    ensure_collection(_vectors)
    yield


app = FastAPI(title="KnowDo AI", lifespan=lifespan)


class AskRequest(BaseModel):
    question: str

    @field_validator("question")
    @classmethod
    def not_blank(cls, value: str) -> str:
        # Rejects whitespace-only questions as a 422 rather than paying
        # for an embedding call on an empty string.
        if not value.strip():
            raise ValueError("question must not be blank")

        return value


class AskResponse(BaseModel):
    answer: str
    sources: list[int]


@app.get("/health")
def health() -> dict:
    return {"status": "ok"}


@app.post("/ask", response_model=AskResponse)
def ask(request: AskRequest) -> dict:
    return answer(request.question.strip(), _vectors)

// Package ai talks to the internal Python AI service.
package ai

import (
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"net/http"
	"time"
)

// ErrUnavailable means the AI service could not be reached or did not
// answer in time — distinct from it answering with an error, because
// only one of those is fixed by starting the container.
var ErrUnavailable = errors.New("ai service unavailable")

// Response is the shape both the Python service and this API return.
type Response struct {
	Answer  string  `json:"answer"`
	Sources []int64 `json:"sources"`
}

// Asker is the boundary the handler depends on, so tests need no server.
type Asker interface {
	Ask(ctx context.Context, question string) (Response, error)
}

type Client struct {
	baseURL string
	http    *http.Client
}

func NewClient(baseURL string) *Client {
	return &Client{
		baseURL: baseURL,
		// Retrieval plus one LLM call. ponytail: a flat timeout — if
		// streaming arrives in a later stage this becomes a stream proxy
		// and the timeout moves to the first byte.
		http: &http.Client{Timeout: 60 * time.Second},
	}
}

func (c *Client) Ask(ctx context.Context, question string) (Response, error) {
	body, err := json.Marshal(map[string]string{"question": question})
	if err != nil {
		return Response{}, err
	}

	req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.baseURL+"/ask", bytes.NewReader(body))
	if err != nil {
		return Response{}, err
	}

	req.Header.Set("Content-Type", "application/json")

	resp, err := c.http.Do(req)
	if err != nil {
		return Response{}, fmt.Errorf("%w: %v", ErrUnavailable, err)
	}

	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return Response{}, fmt.Errorf("ai service returned %s", resp.Status)
	}

	var out Response
	if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
		return Response{}, err
	}

	return out, nil
}

import (
	"encoding/json"

	"log"
	"net/http"

type Handlers struct {
	asker Asker
}

func NewHandlers(asker Asker) *Handlers {
	return &Handlers{asker: asker}
}

func (h *Handlers) HandleAsk(w http.ResponseWriter, r *http.Request) {
	var body struct {
		Question string `json:"question"`
	}

	if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
		http.Error(w, "invalid JSON body", http.StatusBadRequest)
		return
	}

	question := strings.TrimSpace(body.Question)
	if question == "" {
		http.Error(w, "question is required", http.StatusBadRequest)
		return
	}

	answer, err := h.asker.Ask(r.Context(), question)
	if err != nil {
		log.Printf("asking ai service: %v", err)

		if errors.Is(err, ErrUnavailable) {
			http.Error(w, "ai service unavailable", http.StatusServiceUnavailable)
			return
		}

		http.Error(w, "failed to answer question", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	if err := json.NewEncoder(w).Encode(answer); err != nil {
		log.Printf("encoding response: %v", err)
	}
}

Inside the existing `<script>` block, before the final `loadTasks();` call:

```js
    async function loadDocuments() {
      const res = await fetch('/documents');
      const docs = await res.json();
      const list = document.getElementById('documents');

      list.innerHTML = '';

      for (const d of docs) {
        const li = document.createElement('li');
        const detail = d.status === 'failed' ? d.error : `${d.chunk_count} chunks`;
        li.textContent = `${d.filename} — ${d.status} (${detail})`;
        list.appendChild(li);
      }

      // Ingestion is asynchronous, so poll while anything is in flight.
      // ponytail: a 2s poll — swap for SSE if the wait ever feels long.
      if (docs.some((d) => d.status === 'pending' || d.status === 'processing')) {
        setTimeout(loadDocuments, 2000);
      }
    }

    document.getElementById('upload').addEventListener('submit', async (e) => {
      e.preventDefault();

      const res = await fetch('/documents', { method: 'POST', body: new FormData(e.target) });

      if (!res.ok) {
        alert(await res.text());
        return;
      }

      e.target.reset();
      loadDocuments();
    });

    document.getElementById('ask').addEventListener('submit', async (e) => {
      e.preventDefault();

      const answer = document.getElementById('answer');
      answer.textContent = 'Thinking…';

      const res = await fetch('/ask', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ question: e.target.question.value }),
      });

      if (!res.ok) {
        answer.textContent = await res.text();
        return;
      }

      const data = await res.json();
      answer.textContent = data.sources.length
        ? `${data.answer}\n\nSources: documents ${data.sources.join(', ')}`
        : data.answer;
    });

    loadDocuments();
```
